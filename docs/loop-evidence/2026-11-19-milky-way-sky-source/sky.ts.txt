import type { FlightExperiment } from "../../../lib/flight-scene";

// Prepared when the selected artwork changes, before flight drawing. The periodic
// field has no arena boundary. Seed and artwork stay identical across replays.
const SIZE = 3072;
const SEED = 191126;
const BANK_X = 0.37;
const PLANE_Y = 0.394;
// Three art-directed examples for the same Practice departure.
export const PRACTICE_SKY = {
  name: "Practice",
  seed: SEED,
  angle: 10,
  breadth: 1,
  contrast: 1,
  exposure: 1,
};
export const SYSTEM_SKIES = [
  {
    name: "Narrows",
    seed: 72231,
    angle: -9,
    breadth: 0.76,
    contrast: 1.35,
    exposure: 1,
  },
  PRACTICE_SKY,
  {
    name: "Broad reach",
    seed: 58197,
    angle: 24,
    breadth: 1.42,
    contrast: 0.83,
    exposure: 0.82,
  },
];
function hash(x: number, y: number) {
  let h = Math.imul(x + SEED, 374761393) ^ Math.imul(y, 668265263);
  h = Math.imul(h ^ (h >>> 13), 1274126177);
  return ((h ^ (h >>> 16)) >>> 0) / 4294967296;
}
function noise(x: number, y: number, cells: number) {
  const xx = x * cells,
    yy = y * cells;
  const ix = Math.floor(xx),
    iy = Math.floor(yy);
  const fx = xx - ix,
    fy = yy - iy;
  const u = fx * fx * (3 - 2 * fx),
    v = fy * fy * (3 - 2 * fy);
  const sample = (a: number, b: number) =>
    hash((a + cells) % cells, (b + cells) % cells);
  return (
    (sample(ix, iy) * (1 - u) + sample(ix + 1, iy) * u) * (1 - v) +
    (sample(ix, iy + 1) * (1 - u) + sample(ix + 1, iy + 1) * u) * v
  );
}
function gaussian(value: number) {
  return Math.exp(-value * value);
}
function makeField(sky: typeof PRACTICE_SKY) {
  const reference = sky.seed === SEED;
  const phaseX = reference ? 0 : hash(sky.seed, 31);
  const phaseY = reference ? 0 : hash(sky.seed, 32);
  const notchCentre = reference ? 0.006 : (hash(sky.seed, 33) - 0.5) * 0.12;
  const notchWidth = reference ? 0.029 : 0.015 + 0.04 * hash(sky.seed, 34);
  const crestCentre = reference ? -0.018 : (hash(sky.seed, 35) - 0.5) * 0.1;
  return (x: number, y: number) => {
    // Paint along the plane. The flight painter sets the direction of the
    // entire prepared sky, so the reach stays straight across tile boundaries.
    const along = x - BANK_X;
    const longitude = along - Math.round(along);
    const offset = y - PLANE_Y;
    const across = (offset - Math.round(offset)) / sky.breadth;
    const bank = gaussian(longitude / 0.088);
    const silver = gaussian((longitude + 0.25) / 0.076);
    const shoulder = 0.013 + 0.038 * bank + 0.009 * silver;
    if (Math.abs(across) > 5 * shoulder) return { light: 0, warmth: 0 };

    // A dark headland cuts in from above the warm stretch. Farther west,
    // smaller breaks expose a cool shelf along the same galactic plane.
    x += phaseX;
    y += phaseY;
    const broad = noise(x, y, 32) - 0.5;
    const flecks = noise(x + broad * 0.023, y, 112) - 0.5;
    const grit = noise(x, y, 384) - 0.5;
    const relief = broad * 0.024 + flecks * 0.012 + grit * 0.004;
    const edge = across + relief;
    const fringe = gaussian(edge / shoulder) * (0.2 + bank * 0.4);
    const shelf = gaussian((edge - 0.004) / (0.004 + shoulder * 0.27));
    const crest = gaussian((longitude - crestCentre) / 0.045);
    const starlight = fringe + shelf * (0.36 + 1.3 * crest + 0.55 * silver);

    const headland = gaussian((longitude - notchCentre) / notchWidth);
    const dustEdge = -0.013 + 0.031 * headland + 0.006 * broad;
    const cut = gaussian((edge - dustEdge) / (0.003 + 0.014 * headland));
    const inlet = gaussian((edge - dustEdge + 0.013 + broad * 0.024) / 0.006);
    const fragments = noise(x + broad * 0.048, y + flecks * 0.025, 72);
    const light =
      starlight *
      (1 - 0.985 * cut) *
      (1 - 0.85 * bank * inlet) *
      (0.26 + 1.65 * fragments * fragments + flecks * 0.38 + grit * 0.28);
    const warmth = Math.max(
      0,
      Math.min(
        1,
        bank * 0.32 + crest * 0.46 + cut * bank * 0.65 + broad * 0.42,
      ),
    );
    return { light: Math.max(0, light) ** sky.contrast, warmth };
  };
}
export function prepareSky(sky: typeof PRACTICE_SKY) {
  const field = makeField(sky);
  const canvas = document.createElement("canvas");
  canvas.width = canvas.height = SIZE;
  const ctx = canvas.getContext("2d");
  if (!ctx) throw new Error("Sky preparation needs Canvas");
  const small = document.createElement("canvas");
  const resolution = 1536;
  small.width = small.height = resolution;
  const sc = small.getContext("2d");
  if (!sc) throw new Error("Sky preparation needs Canvas");
  const data = sc.createImageData(resolution, resolution);
  // Match 02.2's mean prepared light signal before the unchanged 45% opacity.
  // This is a comparison condition, not a final exposure or colour contract.
  let total = 0;
  for (let y = 0; y < 256; y++)
    for (let x = 0; x < 256; x++) total += field(x / 256, y / 256).light;
  const gain = 0.03618919255854648 / (total / (256 * 256));
  for (let y = 0; y < resolution; y++)
    for (let x = 0; x < resolution; x++) {
      const u = x / resolution,
        v = y / resolution;
      const sample = field(u, v);
      // Roll off the highlights so the brightest bank retains its folds.
      const exposure = sample.light * gain * sky.exposure;
      const f = (2.4 * exposure) / (1.75 + exposure);
      const grain = 0.78 + hash(x, y) * 0.44;
      const warmth = sample.warmth;
      const k = (y * resolution + x) * 4;
      data.data[k] = 3 + f * (86 + warmth * 52) * grain;
      data.data[k + 1] = 5 + f * (88 + warmth * 25) * grain;
      data.data[k + 2] = 10 + f * (109 - warmth * 25) * grain;
      data.data[k + 3] = 255;
    }
  sc.putImageData(data, 0, 0);
  ctx.drawImage(small, 0, 0, SIZE, SIZE);
  // Dense, mostly unresolved stellar grain follows the same broken light.
  for (let i = 0; i < 145000; i++) {
    const u = hash(i, 1),
      v = hash(i, 2),
      f = Math.min(0.9, field(u, v).light * gain);
    if (hash(i, 3) > 0.035 + f * 0.85) continue;
    const bright = hash(i, 4);
    const alpha = 0.1 + bright * bright * (0.23 + f * 0.44);
    const warm = hash(i, 5) > 0.83;
    ctx.fillStyle = warm
      ? `rgba(230,213,186,${alpha})`
      : `rgba(209,222,241,${alpha})`;
    const r = bright > 0.995 ? 1.3 : 0.35 + bright * 0.45;
    ctx.beginPath();
    ctx.arc(u * SIZE, v * SIZE, r, 0, Math.PI * 2);
    ctx.fill();
  }
  const stars = document.createElement("canvas");
  stars.width = stars.height = SIZE;
  const st = stars.getContext("2d");
  if (!st) throw new Error("Star preparation needs Canvas");
  // All the stellar points share the plane's coordinates and travel. The
  // exposure control can dim the unresolved stream independently of them.
  const point = (
    u: number,
    v: number,
    radius: number,
    alpha: number,
    warm: boolean,
  ) => {
    const x = u * SIZE,
      y = v * SIZE;
    const colour = warm ? "245,219,185" : "209,223,244";
    if (radius > 1.8) {
      const halo = st.createRadialGradient(x, y, 0, x, y, radius * 3.2);
      halo.addColorStop(0, `rgba(${colour},0.24)`);
      halo.addColorStop(1, `rgba(${colour},0)`);
      st.fillStyle = halo;
      st.fillRect(
        x - radius * 3.2,
        y - radius * 3.2,
        radius * 6.4,
        radius * 6.4,
      );
    }
    st.fillStyle = `rgba(${colour},${alpha})`;
    st.beginPath();
    st.arc(x, y, radius, 0, Math.PI * 2);
    st.fill();
  };
  // Most of the open field is quiet. More middle-distance points resolve
  // where the light of the galactic plane shows through the dust.
  for (let i = 0; i < 4200; i++) {
    const u = hash(i, 21),
      v = hash(i, 22);
    const sample = field(u, v);
    const density = 0.1 + Math.min(0.75, sample.light * gain * 0.55);
    if (hash(i, 25) > density) continue;
    const rank = hash(i, 23) ** 3;
    point(
      u,
      v,
      0.5 + 1.05 * rank,
      0.24 + 0.43 * rank,
      hash(i, 24) < 0.25 + sample.warmth * 0.35,
    );
  }

  // Unequal intervals let a pair and a solitary star read across the coast.
  // These positions belong to the artwork and pass freely behind local bodies.
  point(0.174, 0.469, 3.0, 0.89, false);
  point(0.183, 0.464, 1.65, 0.73, false);
  point(0.21, 0.449, 1.14, 0.6, true);
  point(0.492, 0.524, 2.55, 0.86, true);
  point(0.542, 0.375, 2.0, 0.8, false);
  point(0.552, 0.388, 1.35, 0.64, false);

  const plane = ctx.createPattern(canvas, "repeat");
  const starfield = st.createPattern(stars, "repeat");
  if (!plane || !starfield)
    throw new Error("Sky preparation needs repeating artwork");
  return { plane, starfield };
}
export function skyExperiment(
  art: ReturnType<typeof prepareSky>,
  pan: number,
  brightness: number,
  showStars: boolean,
  placement: { x: number; y: number; angle: number; zoom: number },
): FlightExperiment {
  // Same starting position as the 02.12 bank crossing.
  const originX = 715;
  const originY = 995;
  return {
    label: "Experimental Milky Way sky",
    create: () => ({
      step() {},
      drawSky(frame, ctx) {
        const { camera } = frame;
        const w = camera.viewportWidth,
          h = camera.viewportHeight;
        const ox = ((camera.pos.x - 960) * pan) / 100 + originX;
        const oy = ((camera.pos.y - 540) * pan) / 100 + originY;
        const pivotX = BANK_X * SIZE;
        const pivotY = PLANE_Y * SIZE;
        const transform = new DOMMatrix()
          .translate(pivotX - ox + placement.x, pivotY - oy + placement.y)
          .rotate(placement.angle)
          .scale(placement.zoom / 100)
          .translate(-pivotX, -pivotY);
        ctx.save();
        art.plane.setTransform(transform);
        ctx.globalAlpha = brightness / 100;
        ctx.fillStyle = art.plane;
        ctx.fillRect(0, 0, w, h);
        if (showStars) {
          ctx.globalAlpha = 1;
          art.starfield.setTransform(transform);
          ctx.fillStyle = art.starfield;
          ctx.fillRect(0, 0, w, h);
        }
        ctx.restore();
      },
    }),
  };
}
