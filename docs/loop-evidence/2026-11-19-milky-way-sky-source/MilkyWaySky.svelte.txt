<script lang="ts">
import { onMount } from "svelte";
import { PRACTICE_FLIGHT_SCENE } from "../../../lib/flight-scene";
import RingSlider from "../../inputs/RingSlider.svelte";
import Switch from "../../inputs/Switch.svelte";
import "@void-drift/core/styles/buttons.css";
import SegmentedControl from "../../inputs/SegmentedControl.svelte";
import FlightSceneInstrument from "../../lab/flight-scene/FlightSceneInstrument.svelte";
import { PRACTICE_SKY, prepareSky, SYSTEM_SKIES, skyExperiment } from "./sky";

let { label }: { label: string } = $props();
let ready = $state(false);
let system = $state("Narrows");
const openingPlacement = { x: 448, y: -144, angle: -32, zoom: 100 };
let placement = $state({ ...openingPlacement });
const systemNames = SYSTEM_SKIES.map((entry) => entry.name);
const sightline = $derived(
  SYSTEM_SKIES.find((entry) => entry.name === system) ?? PRACTICE_SKY,
);
const art = $derived(ready ? prepareSky(sightline) : undefined);
let sky = $state("Milky Way");
let pan = $state(65);
let brightness = $state(45);
let showStars = $state(true);
const starsId = $props.id();
const config = {
  ...PRACTICE_FLIGHT_SCENE,
  opening: { burn: "course", view: "game", tick: 150 },
};
const experiment = $derived(
  art && sky === "Milky Way"
    ? skyExperiment(art, pan, brightness, showStars, { ...placement })
    : undefined,
);
onMount(() => {
  ready = true;
});
</script>
<FlightSceneInstrument
  {config}
  {label}
  {experiment}
  summary={sky === "Milky Way"
    ? `Milky Way · ${system} · offset (${placement.x}, ${placement.y}) stage px · ${placement.angle}° · pan ${pan} stage px / 100 world u · stream ${brightness}% · stars ${showStars ? "on" : "off"} · pattern zoom ${placement.zoom}% · dust seed ${sightline.seed}`
    : "Production sky · world anchored"}
>
  {#snippet pane()}
    <div class="choices">
      <h3>Sky</h3>
      <SegmentedControl
        label="Sky"
        options={["Milky Way", "Production"]}
        bind:value={sky}
        stretch
      />
      {#if sky === "Milky Way"}
        <SegmentedControl
          label="System sightline"
          options={systemNames}
          bind:value={system}
          stretch
        />
        <RingSlider
          label="Horizontal offset"
          unit="px"
          min={-1536}
          max={1536}
          step={8}
          bind:value={placement.x}
        />
        <RingSlider
          label="Vertical offset"
          unit="px"
          min={-1536}
          max={1536}
          step={8}
          bind:value={placement.y}
        />
        <RingSlider
          label="Rotation"
          unit="°"
          min={-180}
          max={180}
          step={1}
          bind:value={placement.angle}
        />
        <RingSlider
          label="Pattern zoom"
          unit="%"
          min={80}
          max={120}
          step={1}
          bind:value={placement.zoom}
        />
        <button
          type="button"
          class="btn btn-normal"
          onclick={() => {
            placement = { x: 0, y: 0, angle: sightline.angle, zoom: 100 };
          }}
        >
          Restore 02.13 placement
        </button>
      {/if}
      <RingSlider
        label="Sky travel"
        unit="px / 100 u"
        min={0}
        max={100}
        step={5}
        bind:value={pan}
      />
      <RingSlider
        label="Stream brightness"
        unit="%"
        min={0}
        max={100}
        step={5}
        bind:value={brightness}
      />
      <div class="star-switch">
        <label for={starsId}>Identifiable stars</label>
        <Switch id={starsId} bind:checked={showStars} />
      </div>
      <p>
        All sightlines use the same Practice flight. Travel is measured in
        logical stage pixels per 100 world units of camera movement.
      </p>
      <button
        type="button"
        class="btn btn-normal"
        onclick={() => {
          sky = "Milky Way";
          system = "Narrows";
          placement = { ...openingPlacement };
          pan = 65;
          brightness = 45;
          showStars = true;
        }}
      >
        Reset sky
      </button>
    </div>
  {/snippet}
</FlightSceneInstrument>

}

}

p {
  margin: 0;
}

}
